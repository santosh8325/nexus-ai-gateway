// Direct Gateway Request Execution (Zero API Key / Backend Routed) async function fetchGroundedGeminiResponse(userQuery, systemPrompt) { // Get the currently selected API key from the dropdown const selectEl = document.getElementById('pg-key-select'); const apiKey = selectEl ? selectEl.value : 'nexus_live_demo123'; const response = await fetch('/api/v1/generate', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + apiKey }, body: JSON.stringify({ prompt: userQuery }) }); const result = await response.json(); if (result.success && result.answer) { return { answer: result.answer, sources: result.sources || [] }; } else { throw new Error(result.error || "Unable to complete request from Nexus AI engine."); } }
